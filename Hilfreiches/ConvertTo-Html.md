# ConvertTo Html

```powershell
$HeadContent = @"
<head>
    <meta charset="UTF-8">
    <style>
        body { font-family: Segoe UI, Arial, Helvetica, sans-serif; background-color: #f4f6f9; margin: 20px; color: #2e2e2e; }
        table { border-collapse: collapse; width: 100%; background-color: #ffffff; }
        th { background-color: #1f4e79; color: #ffffff; padding: 10px; border: 1px solid #d0d7de; }
        td { padding: 8px; border: 1px solid #d0d7de; }
        tr:nth-child(even) { background-color: #f2f2f2; }
        tr:hover { background-color: #dbe9f4; }
    </style>
</head>
"@

$htmladmin = Get-ADGroupMember -Identity Administratoren -Recursive | Select-Object -Property Name,SamAccountName,distinguishedName | ConvertTo-Html -Fragment
$htmlsvc = Get-Service | Select-Object -Property Name,Starttype,Status,Displayname | ConvertTo-Html -Fragment

ConvertTo-Html -Head $HeadContent -Body ("<h1>Server Status</h1>",[string]$htmladmin,"<hr>",[string]$htmlsvc) -Title "ServerStatus" | Out-File -FilePath C:\Testfiles\srvstatus.html
```
